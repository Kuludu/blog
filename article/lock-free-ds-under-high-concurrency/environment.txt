Reproduction (from this directory)
  clang++ -std=c++20 -O3 -DNDEBUG -Wall -Wextra -Wpedantic -pthread benchmark.cpp -o /tmp/stack-benchmark
  /tmp/stack-benchmark --pairs 1000000 --repeats 7 --threads 1,2,4,8,16,32,64,128 > results.csv
  python3 summarize.py results.csv

  clang++ -std=c++20 -O3 -DNDEBUG -DSTACK_COUNT_RETRIES=0 -Wall -Wextra -Wpedantic -pthread benchmark.cpp -o /tmp/stack-benchmark-no-count
